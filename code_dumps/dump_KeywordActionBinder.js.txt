import React, { useState, useEffect } from 'react';

let ipcRenderer;
if (window && window.require) {
  ipcRenderer = window.require('electron').ipcRenderer;
}

function KeywordActionBinder() {
  console.log('[Renderer] KeywordActionBinder mounted');

  const [keyword, setKeyword] = useState('');
  const [category, setCategory] = useState('');
  const [actionType, setActionType] = useState('');
  const [message, setMessage] = useState('');
  const [color, setColor] = useState('#ff0000');
  const [fontSize, setFontSize] = useState(24); // Fixed: was '24px'
  const [duration, setDuration] = useState(5); // In seconds
  const [timerName, setTimerName] = useState('');
  const [timerDirection, setTimerDirection] = useState('down');
  const [timerColor, setTimerColor] = useState('#00ff00');
  const [timerSize, setTimerSize] = useState('medium');
  const [soundFile, setSoundFile] = useState(null);
  const [keyPressText, setKeyPressText] = useState('');
  const [keywordActions, setKeywordActions] = useState([]);
  const [triggerStates, setTriggerStates] = useState({});
  const [categoryStates, setCategoryStates] = useState({});

  useEffect(() => {
    console.log('[KeywordActionBinder] Loading categories from localStorage');
    const savedActions = JSON.parse(localStorage.getItem('keywordActions') || '[]');
    const savedCategoryStates = JSON.parse(localStorage.getItem('categoryStates') || '{}');
    const savedTriggerStates = JSON.parse(localStorage.getItem('triggerStates') || '{}');
    setKeywordActions(savedActions);
    setCategoryStates(savedCategoryStates);
    setTriggerStates(savedTriggerStates);
  }, []);

  useEffect(() => {
    localStorage.setItem('keywordActions', JSON.stringify(keywordActions));
  }, [keywordActions]);

  useEffect(() => {
    localStorage.setItem('categoryStates', JSON.stringify(categoryStates));
  }, [categoryStates]);

  useEffect(() => {
    localStorage.setItem('triggerStates', JSON.stringify(triggerStates));
  }, [triggerStates]);

  const handleAddAction = () => {
    if (!keyword || !actionType) return;

    let action = { type: actionType };

    if (actionType === 'Play Sound') {
      if (!soundFile) return;
      action.file = soundFile;
    } else if (actionType === 'Show Text Alert') {
      action.message = message;
      action.color = color;
      action.fontSize = fontSize;
      action.duration = duration * 1000; // FIXED: convert to ms
    } else if (actionType === 'Start Timer') {
      action.name = timerName;
      action.duration = parseInt(duration);
      action.direction = timerDirection === 'up' ? 'up' : 'down';
      action.color = timerColor;
      action.size = timerSize;
    } else if (actionType === 'Key Press') {
      action.keys = keyPressText;
    }

    const newAction = { keyword, category: category || 'default', action };
    const id = `${keyword}_${category}`;
    setKeywordActions([...keywordActions, newAction]);
    setTriggerStates({ ...triggerStates, [id]: true });
    setCategoryStates({ ...categoryStates, [category || 'default']: true });

    setKeyword('');
    setCategory('');
    setMessage('');
    setFontSize(24);
    setDuration(5);
    setTimerName('');
    setSoundFile(null);
    setKeyPressText('');
  };

  const handleDelete = (index) => {
    const updated = [...keywordActions];
    updated.splice(index, 1);
    setKeywordActions(updated);
  };

  const handleSendChat = (msg) => {
    const filtered = keywordActions.filter(({ keyword, category }) => {
      const id = `${keyword}_${category}`;
      return categoryStates[category] && triggerStates[id];
    });
    ipcRenderer.send('chat-message', {
      message: msg,
      keywordActions: filtered
    });
  };

  const toggleCategory = (cat) => {
    setCategoryStates({ ...categoryStates, [cat]: !categoryStates[cat] });
  };

  const toggleTrigger = (keyword, cat) => {
    const id = `${keyword}_${cat}`;
    setTriggerStates({ ...triggerStates, [id]: !triggerStates[id] });
  };

  return (
    <div>
      <h2>Keyword Action Binder</h2>

      <input
        placeholder="Keyword"
        value={keyword}
        onChange={(e) => setKeyword(e.target.value)}
      />
      <input
        placeholder="Category"
        value={category}
        onChange={(e) => setCategory(e.target.value)}
      />

      <select value={actionType} onChange={(e) => setActionType(e.target.value)}>
        <option value="">Select Action</option>
        <option value="Show Text Alert">Show Text Alert</option>
        <option value="Play Sound">Play Sound</option>
        <option value="Start Timer">Start Timer</option>
        <option value="Key Press">Key Press</option>
      </select>

      {actionType === 'Show Text Alert' && (
        <div>
          <input
            placeholder="Message"
            value={message}
            onChange={(e) => setMessage(e.target.value)}
          />
          <input
            type="color"
            value={color}
            onChange={(e) => setColor(e.target.value)}
          />
          <input
            type="number"
            value={fontSize}
            onChange={(e) => setFontSize(Number(e.target.value))}
          />
          <input
            type="number"
            value={duration}
            onChange={(e) => setDuration(Number(e.target.value))}
          />
        </div>
      )}

      {actionType === 'Start Timer' && (
        <div>
          <input
            placeholder="Name"
            value={timerName}
            onChange={(e) => setTimerName(e.target.value)}
          />
          <select
            value={timerDirection}
            onChange={(e) => setTimerDirection(e.target.value)}
          >
            <option value="down">Count Down</option>
            <option value="up">Count Up</option>
          </select>
          <input
            type="color"
            value={timerColor}
            onChange={(e) => setTimerColor(e.target.value)}
          />
          <select
            value={timerSize}
            onChange={(e) => setTimerSize(e.target.value)}
          >
            <option value="small">Small</option>
            <option value="medium">Medium</option>
            <option value="large">Large</option>
          </select>
        </div>
      )}

      {actionType === 'Key Press' && (
        <input
          placeholder="Keys to Press"
          value={keyPressText}
          onChange={(e) => setKeyPressText(e.target.value)}
        />
      )}

      <button onClick={handleAddAction}>Add Action</button>

      <hr />
      <input
        placeholder="Chat Message"
        onKeyDown={(e) => {
          if (e.key === 'Enter') handleSendChat(e.target.value);
        }}
      />
      <button onClick={() => handleSendChat(document.querySelector('input[placeholder="Chat Message"]').value)}>
        Send Chat Message
      </button>

      <hr />
      {Object.keys(categoryStates).map((cat) => (
        <div key={cat}>
          <h3>
            <input
              type="checkbox"
              checked={categoryStates[cat]}
              onChange={() => toggleCategory(cat)}
            />{' '}
            {cat}
          </h3>
          {keywordActions
            .filter((a) => a.category === cat)
            .map((a, index) => (
              <div key={index}>
                <span>
                  <input
                    type="checkbox"
                    checked={triggerStates[`${a.keyword}_${a.category}`]}
                    onChange={() => toggleTrigger(a.keyword, a.category)}
                  />{' '}
                  {a.keyword} â†’ {a.action.type}
                </span>
                <button onClick={() => handleDelete(index)}>Delete</button>
              </div>
            ))}
        </div>
      ))}
    </div>
  );
}

export default KeywordActionBinder;
