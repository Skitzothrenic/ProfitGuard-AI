import React, { useEffect, useState } from 'react';
import KeywordActionBinder from './components/KeywordActionBinder';
import TimerDisplay from './components/TimerDisplay';
import AlertBanner from './components/AlertBanner';

function App() {
  console.log('[App] Mounting');

  const [timers, setTimers] = useState([]);
  const [alerts, setAlerts] = useState([]);
  const [keywordActions, setKeywordActions] = useState([]);
  const [chatInput, setChatInput] = useState('');
  const [categories, setCategories] = useState({});
  const [triggerStates, setTriggerStates] = useState({});

  useEffect(() => {
    const savedCats = localStorage.getItem('keywordCategories');
    if (savedCats) setCategories(JSON.parse(savedCats));

    const savedTriggers = localStorage.getItem('triggerStates');
    if (savedTriggers) setTriggerStates(JSON.parse(savedTriggers));

    if (window.electronAPI) {
      console.log('[App] Registering sound and alert listeners');

      window.electronAPI.onPlaySound((event, soundPathOrBase64) => {
        console.log('[App] Received play-sound from main:', soundPathOrBase64);
        try {
          const audio = new Audio(soundPathOrBase64);
          audio.play().catch(err => console.error('[App] Sound playback failed:', err));
        } catch (error) {
          console.error('[App] Invalid audio source:', error);
        }
      });

      window.electronAPI.onShowAlert((config) => {
        console.log('[App] Received show-alert from main:', config);
        const {
          message = 'Alert',
          color = '#f00',
          duration = 3,
          fontSize = 24
        } = config;

        setAlerts(prev => [
          ...prev,
          {
            id: Date.now(),
            message,
            duration: duration * 1000, // âœ… Convert seconds to ms
            color,
            fontSize
          }
        ]);
      });

      window.electronAPI.onStartTimer((config) => {
        console.log('[App] Received start-timer from main:', config);
        if (!config || typeof config !== 'object') {
          console.error('[App] Invalid timer config received:', config);
          return;
        }

        const {
          name = 'Unnamed',
          duration = 10,
          direction = 'down',
          color = '#00ff00',
          size = 'medium',
        } = config;

        setTimers(prev => [...prev, {
          id: Date.now(),
          name,
          duration,
          direction,
          color,
          size
        }]);
      });
    }
  }, []);

  const removeTimer = (id) => setTimers(prev => prev.filter(t => t.id !== id));
  const removeAlert = (id) => setAlerts(prev => prev.filter(a => a.id !== id));
  const handleAddKeywordAction = (newAction) => setKeywordActions(prev => [...prev, newAction]);

  const updateCategoryState = (newCategories) => {
    setCategories(newCategories);
    localStorage.setItem('keywordCategories', JSON.stringify(newCategories));
  };

  const updateTriggerState = (newTriggerStates) => {
    setTriggerStates(newTriggerStates);
    localStorage.setItem('triggerStates', JSON.stringify(newTriggerStates));
  };

  const getFilteredActions = () => {
    return keywordActions.filter(entry => {
      const cat = entry.category || 'Uncategorized';
      const id = `${entry.keyword}_${cat}`;
      const catEnabled = categories[cat]?.enabled !== false;
      const trigEnabled = triggerStates[id]?.enabled !== false;
      console.log(`[App][Filter Check] keyword="${entry.keyword}", category="${cat}", categoryEnabled=${catEnabled}, triggerEnabled=${trigEnabled}`);
      return catEnabled && trigEnabled;
    });
  };

  const handleSendChat = () => {
    console.log('[App] Send Chat Message clicked.');
    console.log('[App] Current keywordActions:', keywordActions);
    console.log('[App] Current category states:', categories);
    console.log('[App] Current trigger states:', triggerStates);

    const filtered = getFilteredActions();
    console.log('[App] Filtered keywordActions to send:', filtered);

    const asyncTasks = [];
    const immediateActions = [];

    filtered.forEach(({ keyword, action, category }) => {
      const cleanAction = { ...action };
      if (cleanAction.file instanceof File) {
        const task = new Promise(resolve => {
          const reader = new FileReader();
          reader.onload = () => {
            cleanAction.base64 = reader.result;
            delete cleanAction.file;
            resolve({ keyword, action: cleanAction });
          };
          reader.readAsDataURL(cleanAction.file);
        });
        asyncTasks.push(task);
      } else {
        immediateActions.push({ keyword, action: cleanAction });
      }
    });

    Promise.all(asyncTasks).then(results => {
      const finalActions = [...immediateActions, ...results];
      const data = {
        message: chatInput,
        keywordActions: finalActions,
      };
      console.log('[App] FINAL chat message payload sent to main:', data);
      window.electronAPI.sendChatMessage(data);
    });
  };

  return (
    <div style={{ padding: '1rem', background: '#1a1a1a', color: '#fff' }}>
      <KeywordActionBinder
        onAddKeywordAction={handleAddKeywordAction}
        keywordActions={keywordActions}
        chatInput={chatInput}
        setChatInput={setChatInput}
        onSendChat={handleSendChat}
        categories={categories}
        setCategories={updateCategoryState}
        triggerStates={triggerStates}
        setTriggerStates={updateTriggerState}
      />

      {alerts.map(alert => (
        <AlertBanner
          key={alert.id}
          message={alert.message}
          duration={alert.duration}
          color={alert.color}
          fontSize={alert.fontSize}
          onClose={() => removeAlert(alert.id)}
        />
      ))}

      {timers.map(timer => (
        <TimerDisplay
          key={timer.id}
          name={timer.name}
          duration={timer.duration}
          direction={timer.direction}
          size={timer.size}
          onFinish={() => removeTimer(timer.id)}
        />
      ))}
    </div>
  );
}

export default App;
